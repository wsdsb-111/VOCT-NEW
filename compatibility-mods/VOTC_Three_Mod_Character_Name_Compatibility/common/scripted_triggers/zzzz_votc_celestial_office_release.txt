# Keep Chinese Celestial repair inside the target's de-jure region.
# Other OE repair callers retain their original title filters.
oe_stream_patch_attachable_trigger = {
	is_title_created = yes
	tier >= tier_county
	is_landless_type_title = no
	is_noble_family_title = no
	is_leased_out = no
	tier < scope:oe_stream_fix_target.highest_held_title_tier
	OR = {
		AND = {
			scope:oe_stream_fix_target = {
				top_liege = {
					has_primary_title = title:h_china
					government_has_flag = government_is_celestial
				}
			}
			has_title_law_flag = appointment_type_succession
			NOT = { has_title_law = SanGong_zongfan_succession_law }
			is_oe_noble_family_title = no
			scope:oe_stream_fix_target = { save_temporary_scope_as = votc_celestial_region_holder }
			votc_celestial_title_in_holder_region_trigger = yes
		}
		NOT = {
			scope:oe_stream_fix_target = {
				top_liege = {
					has_primary_title = title:h_china
					government_has_flag = government_is_celestial
				}
			}
		}
	}
}

votc_celestial_office_region_character_trigger = {
	is_alive = yes
	is_landed = yes
	is_independent_ruler = no
	top_liege = {
		is_alive = yes
		is_landed = yes
		government_has_flag = government_is_celestial
		has_primary_title = title:h_china
	}
}

	votc_celestial_office_region_character_trigger = yes

}

votc_celestial_office_lateral_gain_trigger = {
	exists = scope:title
	scope:title = {
		holder = root
		tier >= tier_county
		tier <= tier_empire
		votc_celestial_office_region_title_trigger = yes
		is_landless_type_title = no
		is_noble_family_title = no
		is_leased_out = no
	}
	root = {
		votc_celestial_office_region_character_trigger = yes
		any_held_title = {
			this != scope:title
			tier = scope:title.tier
			votc_celestial_office_region_title_trigger = yes
			is_landless_type_title = no
			is_noble_family_title = no
			is_leased_out = no
		}
	}
}

votc_celestial_title_in_holder_region_trigger = {
	save_temporary_scope_as = votc_celestial_region_checked_title
	OR = {
		target_is_de_jure_liege_or_above = scope:votc_celestial_region_holder.primary_title
		scope:votc_celestial_region_holder = {
			any_held_title = {
				this != scope:votc_celestial_region_checked_title
				tier > scope:votc_celestial_region_checked_title.tier
				is_landless_type_title = no
				is_noble_family_title = no
				is_oe_noble_family_title = no
				is_leased_out = no
				scope:votc_celestial_region_checked_title = {
					target_is_de_jure_liege_or_above = prev
				}
			}
		}
	}
}

votc_celestial_office_release_heir_trigger = {
	this != scope:votc_celestial_cleanup_root
	is_alive = yes
	OR = {
		AND = {
			is_landed = yes
			top_liege = scope:votc_celestial_cleanup_emperor
			highest_held_title_tier < scope:votc_celestial_release_title.tier
		}
		AND = {
			is_landed = no
			OR = {
				top_liege = scope:votc_celestial_cleanup_emperor
				is_courtier_of = scope:votc_celestial_cleanup_root
				is_courtier_of = scope:votc_celestial_cleanup_emperor
				employer = {
					OR = {
						this = scope:votc_celestial_cleanup_emperor
						top_liege = scope:votc_celestial_cleanup_emperor
					}
				}
			}
		}
	}
}
